Jeżeli moduł nie jest zdolny się podłączyć do bazy danych, to prawdopodobnie to dlatego, że nie macie zainstalowanego MySQL server.
Niestety, na razie nie mamy dedykowanego servera, więc musimy się bawić tym, co mamy.

Należy pobrać MySQL ze strony: https://dev.mysql.com/downloads/file/?id=536355

następnie zainstalować go z domyślnymi ustawieniami oraz 
login: root
hasło: root

(możecie to zauważyć nawet w kodzie)

jeśli nadal nie działa, to trzeba dodać bazy danych wraz z ich tabelami 
(OBOWIĄZKOWO pilnować rozmiar wpisywanych literek w nazwy BD i tabeli):

create database resourceBase

use resourceBase

create table resources (
hashcode_id INT PRIMARY KEY,
resource JSON NOT NULL
)



create database userDatabase

use userDatabase

create table users (
email VARCHAR(50) PRIMARY KEY,
user JSON NOT NULL,
encryptionKey VARCHAR(256),
password VARCHAR(256),
emailVerified BOOL,
token VARCHAR(256)
)


create database volunteerDatabase;

use volunteerDatabase;

CREATE TABLE organisations (
    OrganisationID INT PRIMARY KEY,
    OrganisationName VARCHAR(255),
    PhoneNumber VARCHAR(20),
    Address VARCHAR(255)
);

CREATE TABLE volunteers (
    VolunteerID INT PRIMARY KEY,
    FirstName VARCHAR(100),
    LastName VARCHAR(100),
    Email VARCHAR(100),
    Gender VARCHAR(10),
    PhoneNumber VARCHAR(20),
    Address VARCHAR(255),
    Experience TEXT,
    AdditionalInfo TEXT,
    Skills TEXT,
    Availability TEXT,
    OrganisationID INT,
    FOREIGN KEY (OrganisationID) REFERENCES Organisations(OrganisationID)
);
