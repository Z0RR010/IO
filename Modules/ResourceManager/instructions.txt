Poprostu informacja dla potencjalnego rozszerzenia i/lub modyfikacji BD

create database resourceBase

use resourceBase

create table resources (
hashcode_id INT PRIMARY KEY,
resource JSON NOT NULL,
email VARCHAR(50)
)



create database userDatabase

use userDatabase

create table users (
email VARCHAR(50) PRIMARY KEY,
user JSON NOT NULL,
encryptionKey VARCHAR(256),
password VARCHAR(256),
emailVerified BOOL,
token VARCHAR(256)
)


create database volunteerDatabase;

use volunteerDatabase;

CREATE TABLE organisations (
    OrganisationID INT PRIMARY KEY,
    OrganisationName VARCHAR(255),
    PhoneNumber VARCHAR(20),
    Address VARCHAR(255)
);

CREATE TABLE volunteers (
    VolunteerID INT PRIMARY KEY,
    FirstName VARCHAR(100),
    LastName VARCHAR(100),
    Email VARCHAR(100),
    Gender VARCHAR(10),
    PhoneNumber VARCHAR(20),
    Address VARCHAR(255),
    Experience TEXT,
    AdditionalInfo TEXT,
    Skills TEXT,
    Availability TEXT,
    OrganisationID INT,
    FOREIGN KEY (OrganisationID) REFERENCES Organisations(OrganisationID)
);


do sprawdzenia zawartości:
select * from nazwa_tablicy

usunięcie:
	treści tablicy: delete from nazwa_tablicy
	samej tablicy: drop table nazwa
	bazy danych: drop database nazwa
	
rozszerzanie pewnej tabilcy:
    alter table nazwa_tablicy
    add nazwa_nowej_kolumny typ
usunięcie kolumny:    
    alter table nazwa_tablicy
    drop column nazwa_kolumny
